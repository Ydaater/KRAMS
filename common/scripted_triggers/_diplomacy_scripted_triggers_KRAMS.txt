
DIPLOMACY_SEND_VOLUNTEERS_ENABLE_TRIGGER = {

    # Block AI to AI volunteers
    if = {
        limit = {
            is_ai = yes
            FROM = { is_ai = yes }
        }
        always = no
    }

    if = {
        limit = {
            check_variable = {
                FROM.modifier@cannot_receive_volunteers > 0
            }
        }
        custom_override_tooltip = {
            tooltip = cannot_receive_volunteers_warning
            always = no
        }
    }
    else_if = {
        limit = {
            OR = {
                has_game_rule = { rule = allow_volunteers option = LIMITED }
                has_game_rule = { rule = allow_volunteers option = ALWAYS_FREE }
            }
        }
        always = yes
    }
    else_if = {
        limit = {
            has_game_rule = {
                rule = allow_volunteers
                option = SAME_IDEOLOGY
            }
        }
        custom_override_tooltip = {
            tooltip = RULE_VOLUNTEERS_SAME_IDEOLOGY_TT
            FROM = { same_government_type_as_ROOT = yes }
        }
    }
    else_if = {
        limit = {
            has_game_rule = {
                rule = allow_volunteers
                option = BLOCKED
            }
        }
        custom_override_tooltip = {
            tooltip = RULE_VOLUNTEERS_BLOCKED_TT
            always = no
        }
    }
}
